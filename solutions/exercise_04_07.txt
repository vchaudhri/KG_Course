LOAD CSV WITH HEADERS FROM 'file:///finkg/wikidata_snp100_company_data.csv' AS row
MERGE (c:Entity:Company {wikidata_id: row.company})
SET c.name = row.companyLabel;


LOAD CSV WITH HEADERS FROM 'file:///finkg/wikidata_snp100_company_data.csv' AS row
MERGE (c:Entity:Company {wikidata_id: row.company})
FOREACH (_ IN CASE
    WHEN row.property ENDS WITH '/P169'
    THEN [1]
    ELSE []
END |
    MERGE (p:Entity:Person {wikidata_id: row.value})
    SET p.name = row.valueLabel
    MERGE (c)-[r:CEO]->(p)
    SET r.wikidata_property = row.property
)
